                     console.assert(masterTemplate, "Missing #BL!");

                     for(let section of BL.index.sectionNodes){

                     indexCreate:                  function (masterTemplate){
                     let idText       = {};  // idText[id] = "html string";
                     let sectionNodes = [];
                     let codeObjects  = {};
                     let childNodes   = masterTemplate.childNodes;
                     let section      = null;
                     // for each section
                     for(let childNode of childNodes){
                         // if it's our table of ID/descriptions
                         if(childNode.tagName == "UL"){
                             for(let listItem of childNode.childNodes){
                                 if(listItem.tagName == "LI"){
                                     idText[listItem.id] = listItem.innerHTML;
                                 }
                             }
                         }
                         // else, if it's a section
                         else if((section = BL.indexSection(childNode)) != null){
                             sectionNodes.push(section);
                             if(section.code){
                                 let node         = section.code;
                                 let codeObjectId = BL.getCodeId(node);

                                 if(codeObjects.hasOwnProperty(codeObjectId))
                                     codeObjects[codeObjectId].push(section.code);
                                 else
                                     codeObjects[codeObjectId] = new Array(section.code);
                                 if(section.codeName)
                                     idText[codeObjectId] = section.codeName.innerHTML.trim();
                             }
                         }
                     }
                     let result = {
                         masterTemplate: masterTemplate,
                         sectionNodes:   sectionNodes,
                         codeObjects:    codeObjects,
                         idText:         idText,
                         tocDiv:         document.getElementById("BL-Toc"),
                         sourceDiv:      document.getElementById("BL-Source")
                     };
                     return BL.index = result;
                 }
                ,
                     indexSection:                  function indexSection(node){
                     let result   = null;  // assume failure
                     if(node.tagName == "SECTION"){
                         let child = node.firstElementChild;
                         // ignore inexplicably empty section
                         if(child){
                             result = {node:node, comment:null, code:null, codeName:null};

                             for(; child; child=child.nextSibling)
                                 if(child.tagName == "DIV"){
                                     if(result.comment == null)
                                         result.comment = child;
                                     else
                                         console.log("Ignoring unexpected div in section.");
                                 }else if(child.tagName == "SCRIPT" || child.tagName == "STYLE"){
                                     if(result.code == null){
                                         result.code = child;
                                         result.codeType=child.tagName;
                                     }else
                                     console.log("ignoring unexpected script in section.");
                                 }else if(child.tagName == "HEADER"){
                                     result.codeName = child;
                                 }
                         }
                     }
                     return result;
                 }
                ,
                     getCodeId:                  function (codeNode){
                     // treat missing 'data-BL' same as blank (root section).
                     let result = " ";
                     if(codeNode.hasAttribute("data-BL"))
                         result = codeNode.getAttribute("data-BL");
                     return result;
                 }

                     sectionAppend:                  function (sourceDiv, section){
                     let outerDiv = document.createElement("div");
                     outerDiv.className = "BL-Section";

                         let commentDiv = section.comment.cloneNode(true);

                         BL.tocAppend(section);
                     }
                     if(section.code){
                         let codeDiv = document.createElement("div");
                         codeDiv.className = "BL-Code";
                         let id = BL.getCodeId(section.code);
                         console.log("id = '" + id + "'");
                         BL.createCodeNameHtml(codeDiv, id);
                         BL.textAppend(codeDiv, section.code.innerHTML);
                         outerDiv.appendChild(codeDiv);
                     }
                     sourceDiv.appendChild(outerDiv);
                 }

                     tocAppend:                  function(section){
                     let commentDiv = section.comment;
                     if(commentDiv){
                         // will append first <h*> tag (if any) to TOC.
                         for(let childNode of commentDiv.children)
                             if(childNode.tagName.match(/^[Hh][0-9]$/)){
                                 let anchor = BL.cloneRename(childNode, "a");
                                 anchor.href = "#" + section.node.id;
                                 let headerClone = childNode.cloneNode(false);
                                 headerClone.appendChild(anchor);
                                 BL.index.tocDiv.appendChild(headerClone);
                                 break;
                             }
                     }
                 }

                     cloneRename:                  function(node, newName){
                     let newParent = document.createElement(newName);
                     for(let childNode of node.childNodes){
                         newParent.appendChild(childNode.cloneNode(true));
                     }
                     return newParent;
                 }

                     createCodeNameHtml:                  function (parentDiv, codeId){
                     //                 console.log("createCodeNameHtml: '" + codeId + "'");
                     if(codeId != " "){
                         let arity = BL.index.codeObjects[codeId].length;
                         let suffix= "&#x3009;"; // 
                         if(arity > 1)
                             suffix += "&#x2795;";
                         suffix += "&#x2630;";
                         let para  = document.createElement("p");
                         let span  = BL.createSpanHtml("BL-CodeNameBrackets", "&#x3008;");
                         para.appendChild(span);
                         span = BL.createSpanHtml("BL-CodeName", BL.index.idText[codeId]);
                         para.appendChild(span);
                         span  = BL.createSpanHtml("BL-CodeNameBrackets", suffix);
                         para.appendChild(span);
                         parentDiv.appendChild(para);
                     }
                 }

                     result.className = className;
                     result.innerHTML = innerHTML;
                     return result;
                 }

                     textAppend:                  function BL_textAppend(parentDiv, text){
                     let indent = null;
                     let lex    = BL.tokenizer(text);
                     for(let token of lex){
                         // if first non-empty line, establish amount of input over-indent.
                         if(!indent && token != '\n'){
                             indent = token.slice(0, token.search(/\S|$/));
                             indent = new RegExp("^" + indent);
                         }

                         if(indent)
                             token = token.replace(indent, "");
                         let element = null;
                         if(token == '\n')
                             element = document.createElement("br");
                         else if(0 == token.search(/^[@]<.*@>$/)
                                 && token != "@" + "<@template@>"){
                             let id=token.slice(2,-2);
                             element = document.createElement("span");
                             element.className = "BL-Code-Ref";
                             element.innerHTML = '\u00AB' + BL.index.idText[id] + '\u00BB';
                         }
                         else if(0 == token.search(/^#[0-9A-Fa-f]+;/)){
                             element = document.createElement("span");
                             element.innerHTML = token;
                             let cssColor = BL.getCssColor(token);
                             element.style.cssText = cssColor;
                         }
                         else
                             element = document.createTextNode(token);
                         parentDiv.appendChild(element);
                     }
                 }
                ,
                     tokenizer:                  function * BL_tokenizer(text){
                     let scrap="";
                     let curPos=0, firstPos=0, maxPos = text.length;
                     if(text.charAt(0) == '\n')
                         curPos = firstPos = 1;
                     for(; curPos < text.length; ++curPos){
                         switch(text.charAt(curPos)){
                             case '\n':
                                 if(curPos > firstPos)
                                     yield text.slice(firstPos, curPos);
                                 yield "\n";
                                 break;
                             case '&': //??? unaware what context we're in, I think
                                 let pattern = /&lt;(\/script|script )/;
                                 scrap = text.slice(curPos, curPos+11);
                                 //  console.log("---scrap = '" + scrap + "'");
                                 if(scrap.match(pattern)){
                                     if(curPos > firstPos)
                                         yield text.slice(firstPos, curPos);
                                     console.log("rewrite: '" + "<" + scrap.slice(4) + "'");
                                     yield "<" + scrap.slice(4);
                                     curPos += (11-1);
                                     break;
                                 }
                                 // else, accumulate
                                 continue;
                                 break;
                             case '#' :
                                 // might be HTML color
                                 scrap = text.slice(curPos, curPos+2+6);
                                 let match = scrap.match(/^#[0-9A-Fa-f]+;/);
                                 if(match && (match[0].length == 5 || match[0].length == 8)){
                                     if(curPos > firstPos)
                                         yield text.slice(firstPos, curPos);
                                     yield match[0];
                                     curPos += (match[0].length - 1);
                                     break;
                                 }
                                 // else, accumulate
                                 continue;
                             case '@' :
                                 // might be code reference ??? need more rigorous match!
                                 if(text.charAt(curPos+1) == '<'){
                                     let matchPos = text.slice(curPos+1).indexOf("@>");
                                     if(matchPos != -1){
                                         // OK, we have a reference
                                         if(curPos > firstPos)
                                             yield text.slice(firstPos, curPos);
                                         yield text.slice(curPos, curPos+matchPos+3);
                                         curPos += matchPos+2;
                                         break;
                                     }
                                 }
                                 // else, accumulate
                                 continue;
                             default: // accumulate
                                 continue;
                         }
                         firstPos = curPos+1;
                     }
                     // if we've accumulated text we haven't emitted
                     if(curPos > firstPos)
                         yield text.slice(firstPos, curPos);
                 }
                ,
                     getCssColor:                  function(token){
                     let r, g, b, match, result = "background-color: " + token;
                     if(token.length == 5){
                         match = token.match(/^#([0-9A-Fa-f])([0-9A-Fa-f])([0-9A-Fa-f]);$/);
                         r = match[1] + match[1];
                         g = match[2] + match[2];
                         b = match[3] + match[3];
                     }
                     else{
                         match = token.match(/^#([0-9A-Fa-f][0-9A-Fa-f])([0-9A-Fa-f][0-9A-Fa-f])([0-9A-Fa-f][0-9A-Fa-f]);$/);
                         r = match[1];
                         g = match[2];
                         b = match[3];
                     }
                     r = parseInt(r,16); g = parseInt(g,16); b = parseInt(b,16);
                     let brightness = (r + r + b + g + g + g)/6;
                     if(brightness > 100){
                         result += " color : black";
                     }
                     else
                         result += " color : white";
                     return result;
                 }

                 }

                    <p>

                 function (masterTemplate){
                     let idText       = {};  // idText[id] = "html string";
                     let sectionNodes = [];
                     let codeObjects  = {};
                     let childNodes   = masterTemplate.childNodes;
                     let section      = null;
                     // for each section
                     for(let childNode of childNodes){
                         // if it's our table of ID/descriptions
                         if(childNode.tagName == "UL"){
                             for(let listItem of childNode.childNodes){
                                 if(listItem.tagName == "LI"){
                                     idText[listItem.id] = listItem.innerHTML;

                                 let node         = section.code;
                                 let codeObjectId = BL.getCodeId(node);

                                 if(codeObjects.hasOwnProperty(codeObjectId))
                                     codeObjects[codeObjectId].push(section.code);
                                 else
                                     codeObjects[codeObjectId] = new Array(section.code);
                                 if(section.codeName)
                                     idText[codeObjectId] = section.codeName.innerHTML.trim();
                             }
                         }
                     }
                     let result = {
                         masterTemplate: masterTemplate,
                         sectionNodes:   sectionNodes,
                         codeObjects:    codeObjects,
                         idText:         idText,
                         tocDiv:         document.getElementById("BL-Toc"),

                 function indexSection(node){
                     let result   = null;  // assume failure
                     if(node.tagName == "SECTION"){
                         let child = node.firstElementChild;
                         // ignore inexplicably empty section
                         if(child){
                             result = {node:node, comment:null, code:null, codeName:null};

                             for(; child; child=child.nextSibling)
                                 if(child.tagName == "DIV"){
                                     if(result.comment == null)
                                         result.comment = child;
                                     else
                                         console.log("Ignoring unexpected div in section.");
                                 }else if(child.tagName == "SCRIPT" || child.tagName == "STYLE"){
                                     if(result.code == null){
                                         result.code = child;
                                         result.codeType=child.tagName;
                                     }else
                                     console.log("ignoring unexpected script in section.");
                                 }else if(child.tagName == "HEADER"){
                                     result.codeName = child;
                                 }
                         }
                     }
                     return result;
                 }

                 function BL_textAppend(parentDiv, text){
                     let indent = null;
                     let lex    = BL.tokenizer(text);
                     for(let token of lex){
                         // if first non-empty line, establish amount of input over-indent.
                         if(!indent && token != '\n'){
                             indent = token.slice(0, token.search(/\S|$/));
                             indent = new RegExp("^" + indent);
                         }

                         if(indent)
                             token = token.replace(indent, "");
                         let element = null;
                         if(token == '\n')
                             element = document.createElement("br");
                         else if(0 == token.search(/^[@]<.*@>$/)
                                 && token != "@" + "<@template@>"){
                             let id=token.slice(2,-2);
                             element = document.createElement("span");
                             element.className = "BL-Code-Ref";
                             element.innerHTML = '\u00AB' + BL.index.idText[id] + '\u00BB';
                         }
                         else if(0 == token.search(/^#[0-9A-Fa-f]+;/)){
                             element = document.createElement("span");
                             element.innerHTML = token;
                             let cssColor = BL.getCssColor(token);
                             element.style.cssText = cssColor;
                         }
                         else
                             element = document.createTextNode(token);
                         parentDiv.appendChild(element);
                     }
                 }

                 function * BL_tokenizer(text){
                     let scrap="";
                     let curPos=0, firstPos=0, maxPos = text.length;
                     if(text.charAt(0) == '\n')
                         curPos = firstPos = 1;
                     for(; curPos < text.length; ++curPos){
                         switch(text.charAt(curPos)){
                             case '\n':
                                 if(curPos > firstPos)
                                     yield text.slice(firstPos, curPos);
                                 yield "\n";
                                 break;
                             case '&': //??? unaware what context we're in, I think
                                 let pattern = /&lt;(\/script|script )/;
                                 scrap = text.slice(curPos, curPos+11);
                                 //  console.log("---scrap = '" + scrap + "'");
                                 if(scrap.match(pattern)){
                                     if(curPos > firstPos)
                                         yield text.slice(firstPos, curPos);
                                     console.log("rewrite: '" + "<" + scrap.slice(4) + "'");
                                     yield "<" + scrap.slice(4);
                                     curPos += (11-1);
                                     break;
                                 }
                                 // else, accumulate
                                 continue;
                                 break;
                             case '#' :
                                 // might be HTML color
                                 scrap = text.slice(curPos, curPos+2+6);
                                 let match = scrap.match(/^#[0-9A-Fa-f]+;/);
                                 if(match && (match[0].length == 5 || match[0].length == 8)){
                                     if(curPos > firstPos)
                                         yield text.slice(firstPos, curPos);
                                     yield match[0];
                                     curPos += (match[0].length - 1);
                                     break;
                                 }
                                 // else, accumulate
                                 continue;
                             case '@' :
                                 // might be code reference ??? need more rigorous match!
                                 if(text.charAt(curPos+1) == '<'){
                                     let matchPos = text.slice(curPos+1).indexOf("@>");
                                     if(matchPos != -1){
                                         // OK, we have a reference
                                         if(curPos > firstPos)
                                             yield text.slice(firstPos, curPos);
                                         yield text.slice(curPos, curPos+matchPos+3);
                                         curPos += matchPos+2;
                                         break;
                                     }
                                 }
                                 // else, accumulate
                                 continue;
                             default: // accumulate
                                 continue;
                         }
                         firstPos = curPos+1;
                     }
                     // if we've accumulated text we haven't emitted
                     if(curPos > firstPos)
                         yield text.slice(firstPos, curPos);
                 }
